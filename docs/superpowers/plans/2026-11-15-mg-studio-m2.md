```js
'use strict';
const { loadTemplate } = require('./narratives');
const DEFAULT_CPS = 4.5;
function estSec(text, cps) { return (String(text || '').length) / (cps || DEFAULT_CPS) + 1.1; }

/** L1 叙事 lint。template 缺省按 story.meta.template 加载。 */
function lintStory({ story, template, charsPerSec = DEFAULT_CPS }) {
  const out = [];
  const t = template || loadTemplate((story && story.meta && story.meta.template) || 'free');
  const beats = (story && story.beats) || [];
  if (!beats.length) return out;
  const roles = beats.map(b => b.role);

  if (t.firstRole && roles[0] !== t.firstRole)
    out.push({ level: 'B', code: 'NARR_HOOK_FIRST', scene: beats[0].id, msg: `首 beat role 应为 ${t.firstRole}，实际 ${roles[0]}` });

  for (const r of (t.requiredRoles || [])) if (!roles.includes(r))
    out.push({ level: 'B', code: 'NARR_MISSING_ROLE', scene: '', msg: `模板 ${t.id} 缺少必需 role: ${r}` });

  if ((t.lastRoles || []).length && !t.lastRoles.includes(roles[roles.length - 1]))
    out.push({ level: 'B', code: 'NARR_END_ROLE', scene: beats[beats.length - 1].id, msg: `末 beat role 应为 ${t.lastRoles.join('/')}，实际 ${roles[roles.length - 1]}` });

  const wantsConflict = (t.requiredRoles || []).concat(t.recommendRoles || []).includes('conflict');
  if (wantsConflict && !roles.includes('conflict'))
    out.push({ level: 'W', code: 'NARR_NO_CONFLICT', scene: '', msg: `模板 ${t.id} 建议有 conflict（常见误解/转折）beat，未检出` });

  const em = beats.map(b => b.emotion).filter(Number.isFinite);
  if (em.length >= 2 && (Math.max(...em) - Math.min(...em)) < 2)
    out.push({ level: 'W', code: 'NARR_FLAT_EMOTION', scene: '', msg: `emotion 极差 ${Math.max(...em) - Math.min(...em)} < 2（情绪平铺）` });

  for (const b of beats) {
    const s = estSec(b.narration, charsPerSec);
    if (s > 15) out.push({ level: 'W', code: 'NARR_LONG_BEAT', scene: b.id, msg: `beat 预估 ${s.toFixed(1)}s > 15s（建议拆分）` });
  }
  return out;
}
module.exports = { lintStory, estSec, DEFAULT_CPS };
```

```js
'use strict';
const { splitSubs, checkSubs } = require('./subs');
const CONNECTIVES = ['所以', '但是', '那么', '接下来', '因此', '而', '于是', '不过', '因为', '首先', '其次', '最后'];
const EXPLAIN = ['就是', '也就是', '一种', '俗称', '指的是', '即', '等于'];
const DEFAULT_CPS = 4.5;

function sentences(text) { return String(text || '').split(/[。！？!?；;\n]/).map(s => s.trim()).filter(Boolean); }

function termUnexplained(text, terms) {
  const out = [];
  for (const term of terms || []) {
    const i = String(text || '').indexOf(term);
    if (i < 0) continue;
    const after = String(text).slice(i + term.length, i + term.length + 20);
    const explained = after.includes('（') || after.includes('(') || EXPLAIN.some(m => after.includes(m));
    if (!explained) out.push(term);
  }
  return out;
}

/** L2 脚本 lint：朗读时长模型/预算/口语/字幕一致性。 */
function lintScript({ story, charsPerSec = DEFAULT_CPS, durationTarget = null, terms = [], subsByBeat = {} }) {
  const out = [];
  const beats = (story && story.beats) || [];
  if (!beats.length) return out;

  let total = 0; const per = [];
  for (const b of beats) { const s = (b.narration || '').length / charsPerSec + 1.1; total += s; per.push({ id: b.id, s }); }
  if (Number.isFinite(durationTarget) && durationTarget > 0) {
    const dev = (total - durationTarget) / durationTarget;
    if (Math.abs(dev) > 0.10) {
      const worst = per.slice().sort((a, b) => b.s - a.s).slice(0, 3).map(x => `${x.id} ${x.s.toFixed(1)}s`).join('、');
      out.push({ level: 'W', code: 'SCRIPT_BUDGET', scene: '', msg: `全片预估 ${total.toFixed(1)}s vs 目标 ${durationTarget}s（偏差 ${(dev * 100).toFixed(0)}%），最长: ${worst}` });
    }
  }

  for (const b of beats) {
    const narr = b.narration || '';
    for (const s of sentences(narr)) if (s.length > 30)
      out.push({ level: 'W', code: 'SCRIPT_LONG_SENTENCE', scene: b.id, msg: `单句 ${s.length} 字 > 30（${s.slice(0, 12)}…）` });
    const tu = termUnexplained(narr, terms);
    if (tu.length) out.push({ level: 'W', code: 'SCRIPT_TERM_UNEXPLAINED', scene: b.id, msg: `术语首现未通俗解释: ${tu.slice(0, 3).join('、')}` });
    const head = narr.slice(0, 12);
    if (!CONNECTIVES.some(c => head.includes(c)))
      out.push({ level: 'W', code: 'SCRIPT_NO_BRIDGE', scene: b.id, msg: 'beat 开头 12 字无衔接词（逻辑桥可能缺失）' });

    const ov = subsByBeat[b.id];
    if (Array.isArray(ov)) {
      const c = checkSubs(ov, narr);
      if (!c.ok) out.push({ level: 'B', code: 'SCRIPT_SUBS_MISMATCH', scene: b.id, msg: c.detail });
    } else if (splitSubs(narr).some(l => l.length > 18)) {
      out.push({ level: 'B', code: 'SCRIPT_SUBS_TOO_LONG', scene: b.id, msg: '自动切句出现 >18 字行' });
    }
  }
  return out;
}
```

```js
'use strict';
const fs = require('fs'), path = require('path');
const FILE = 'calibration.json';
const DEFAULT_VOICE = { local: 'Tingting', bailian: 'longanlingxin' };

function keyOf(backend, voice) {
  const b = backend || 'local';
  return b + ':' + (voice || DEFAULT_VOICE[b] || 'Tingting');
}
function charsPerSecFromPairs(pairs) {
  const chars = (pairs || []).reduce((n, p) => n + (p.chars || 0), 0);
  const dur = (pairs || []).reduce((n, p) => n + (p.dur || 0), 0);
  return dur > 0 ? +(chars / dur).toFixed(3) : null;
}
function readCalibration(dir) {
  try { return JSON.parse(fs.readFileSync(path.join(dir, FILE), 'utf8')); } catch (e) { return {}; }
}
function writeCalibrationSample(dir, key, charsPerSec) {
  const m = readCalibration(dir);
  const prev = m[key] || {};
  m[key] = { charsPerSec, samples: (prev.samples || 0) + 1, updatedAt: new Date().toISOString() };
  fs.writeFileSync(path.join(dir, FILE), JSON.stringify(m, null, 2));
  return m;
}
function charPerSecFor(dir, key, fallback = 4.5) {
  const m = readCalibration(dir);
  return (m[key] && Number.isFinite(m[key].charsPerSec)) ? m[key].charsPerSec : fallback;
}
module.exports = { keyOf, charsPerSecFromPairs, readCalibration, writeCalibrationSample, charPerSecFor, FILE };
```

```js
#!/usr/bin/env node
'use strict';
/* lint_story.js — G0 脚本门（叙事 + 脚本 + 医学三层合成，§9.1）
 * 用法: node lint_story.js [项目目录] [--md]
 * 产物: <项目>/reports/story-lint.md；退出码 0 通过 / 1 有 B 级 / 2 加载失败
 */
const fs = require('fs'), path = require('path');
const ir = require('../lib/ir');
const { LAYOUT_NAMES } = require('../lib/components');
const { loadTemplate } = require('../lib/narratives');
const { lintStory } = require('../lib/story-lint');
const { lintScript } = require('../lib/script-lint');
const { lintMedical } = require('../lib/medical-lint');
const cal = require('../lib/calibrate');

function loadTerms() {
  try { return JSON.parse(fs.readFileSync(path.join(__dirname, '..', 'medical', 'terms.json'), 'utf8')); }
  catch (e) { return []; }
}

function main() {
  const args = process.argv.slice(2);
  const positional = args.filter(a => !a.startsWith('--'));
  const dir = positional.length ? path.resolve(positional[0]) : process.cwd();
  let project;
  try { project = ir.loadProject(dir, LAYOUT_NAMES); }
  catch (e) { console.error(e.message); return 2; }
  if (project.mode === 'v1') { console.log('v1 直通：跳过 G0 脚本门'); return 0; }

  const story = project.story, meta = story.meta || {};
  const template = loadTemplate(meta.template || 'free');
  const key = cal.keyOf(story.tts && story.tts.backend, story.tts && story.tts.voice);
  const cps = cal.charPerSecFor(dir, key, 4.5);
  const subsByBeat = {};
  for (const sc of project.scenes) if (Array.isArray(sc.subsOverride)) subsByBeat[sc.id] = sc.subsOverride;

  const diag = [].concat(
    lintStory({ story, template, charsPerSec: cps }),
    lintScript({ story, charsPerSec: cps, durationTarget: meta.durationTarget, terms: loadTerms(), subsByBeat }),
    lintMedical({ story, scenes: project.scenes })
  );

  const B = diag.filter(d => d.level === 'B'), W = diag.filter(d => d.level === 'W');
  const lines = ['# G0 脚本门报告', '', `项目：${meta.title || dir}（template=${meta.template}，charsPerSec=${cps}）`, `阻断 ${B.length} · 警告 ${W.length}`, ''];
  for (const d of B) lines.push(`- [B] ${d.code} ${d.scene || ''} ${d.msg}`);
  for (const d of W) lines.push(`- [W] ${d.code} ${d.scene || ''} ${d.msg}`);
  const md = lines.join('\n') + '\n';
  fs.mkdirSync(path.join(dir, 'reports'), { recursive: true });
  fs.writeFileSync(path.join(dir, 'reports', 'story-lint.md'), md);

  for (const d of B) console.error(`[B] ${d.code} ${d.scene || ''} ${d.msg}`);
  for (const d of W) console.log(`[W] ${d.code} ${d.scene || ''} ${d.msg}`);
  console.log(`\nG0 ${B.length ? 'BLOCKED' : 'OK'}（B=${B.length} W=${W.length}）→ reports/story-lint.md`);
  return B.length ? 1 : 0;
}
if (require.main === module) process.exit(main());
module.exports = { main };
```

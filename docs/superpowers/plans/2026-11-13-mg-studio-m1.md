**跨任务核心契约（所有任务遵守）：**

- 组件形状：`{name, css(tokens)→string, runtime?(tokens)→string, build(scene)→string, extraUpdate?(scene)→string, boxes(scene)→[{id,x,y,w,h,text?,colorKey?}]}`。build 发射的代码块必须产出 `S['<sceneId>'] = {r, els:[{key,e}]}` 并 `scenesEl.append(r)`；stat 组件额外设 `e.__num`。
- 归一化元素 spec：`{id, type, props, enter:{preset,at,dur}|null, emphasis:[{preset,cue,dur}], exit:{preset,dur}|null}`。
- 运行时全局：`clamp01/lerp/eo/eio/seg/srand/H/op/drawBG/fadeScene/PRE/chan/put/cueLt/upEl/S/TL/SPECS/UPS/OFF`。
- `upEl(e, sp, lt, dur, sc)`：sp 为归一化 spec；sc 为 timeline 场景（含 start/audioStart/dur）。
- `loadProject(dir, layoutNames)` → `{mode:'v2', story, tokens, scenes}`（scenes 按 beats 顺序，已合并 `narration` 与 `beat:{role,emotion,intent,facts}`）或 `{mode:'v1', script}`。
- `compileProject(dir)` → `{mode:'v1'}` 或 `{mode:'v2', scriptPath, htmlPath, lint:[...], errors:[...]}`；有 B 级错误时不写文件。
- `lintMotion({scenes, tokens, components})` → `[{level:'B'|'W', code, scene?, el?, msg}]`；scenes 必须已归一化。

```js
'use strict';
const fs = require('fs'), path = require('path');

const ROLES = ['hook', 'context', 'conflict', 'mechanism', 'evidence', 'demo', 'takeaway', 'cta', 'objective', 'prerequisite', 'steps', 'summary', 'selfcheck'];
const AUDIENCES = ['patient', 'professional', 'general'];
const TEMPLATES = ['science', 'product', 'teaching', 'free'];
const ID_RE = /^[a-z][a-z0-9_]*$/i;
const HEX_RE = /^#[0-9a-f]{6}$/i;
const CUE_RE = /^(narration:[0-9.]+|lt:[0-9.]+)$/;

function validateStory(story) {
  const errors = [];
  if (!story || story.schema !== 2) errors.push('story.schema 必须为 2');
  const m = story && story.meta || {};
  if (!m.title || typeof m.title !== 'string') errors.push('meta.title 缺失');
  if (!AUDIENCES.includes(m.audience)) errors.push('meta.audience 非法: ' + m.audience);
  if (!TEMPLATES.includes(m.template)) errors.push('meta.template 非法: ' + m.template);
  if (!(typeof m.durationTarget === 'number' && m.durationTarget > 0)) errors.push('meta.durationTarget 必须为正数');
  const beats = (story && story.beats) || [];
  if (!Array.isArray(beats) || beats.length === 0) errors.push('beats 不能为空');
  const seen = new Set();
  for (const b of beats) {
    if (!b.id || !ID_RE.test(b.id)) errors.push('beat.id 非法: ' + b.id);
    if (seen.has(b.id)) errors.push('beat.id 重复: ' + b.id);
    seen.add(b.id);
    if (!ROLES.includes(b.role)) errors.push(`beat ${b.id} role 非法: ${b.role}`);
    if (!Number.isInteger(b.emotion) || b.emotion < 1 || b.emotion > 5) errors.push(`beat ${b.id} emotion 须为 1..5`);
    if (typeof b.intent !== 'string') errors.push(`beat ${b.id} intent 须为字符串`);
    if (!b.narration || typeof b.narration !== 'string') errors.push(`beat ${b.id} narration 缺失`);
    if (!Array.isArray(b.facts)) errors.push(`beat ${b.id} facts 须为数组`);
    else for (const f of b.facts) if (typeof f.verified !== 'boolean') errors.push(`beat ${b.id} fact.verified 须为布尔`);
  }
  return { ok: errors.length === 0, errors };
}

function validateScene(sc, beatIds, layoutNames) {
  const errors = [];
  if (!sc || !ID_RE.test(sc.id || '')) errors.push('scene.id 非法: ' + (sc && sc.id));
  if (!beatIds.includes(sc.beat)) errors.push(`scene ${sc.id} 引用不存在的 beat: ${sc.beat}`);
  if (sc.id !== sc.beat) errors.push(`M1 要求 scene.id === beat（${sc.id} vs ${sc.beat}）`);
  if (!layoutNames.includes(sc.layout)) errors.push(`scene ${sc.id} layout 未知: ${sc.layout}`);
  if (!(typeof sc.minSec === 'number' && sc.minSec >= 0)) errors.push(`scene ${sc.id} minSec 须 >= 0`);
  const els = sc.elements || [];
  if (!Array.isArray(els) || els.length === 0) errors.push(`scene ${sc.id} elements 不能为空`);
  const seen = new Set();
  for (const el of els) {
    if (!el.id || seen.has(el.id)) errors.push(`scene ${sc.id} element.id 缺失/重复: ${el.id}`);
    seen.add(el.id);
    if (typeof el.type !== 'string') errors.push(`scene ${sc.id} element ${el.id} 缺 type`);
    if (typeof el.props !== 'object' || el.props == null) errors.push(`scene ${sc.id} element ${el.id} props 须为对象`);
    for (const c of el.emphasis || []) if (!CUE_RE.test(c.cue || '')) errors.push(`scene ${sc.id} element ${el.id} cue 非法: ${c.cue}`);
  }
  for (const c of sc.cues || []) {
    if (!CUE_RE.test(c.at || '')) errors.push(`scene ${sc.id} cue.at 非法: ${c.at}`);
    if (!seen.has(c.target)) errors.push(`scene ${sc.id} cue.target 不存在: ${c.target}`);
  }
  if (sc.subsOverride != null && !Array.isArray(sc.subsOverride)) errors.push(`scene ${sc.id} subsOverride 须为数组或 null`);
  return { ok: errors.length === 0, errors };
}

function validateTokens(tk) {
  const errors = [];
  if (!tk || typeof tk.colors !== 'object') return { ok: false, errors: ['tokens.colors 缺失'] };
```

```js
'use strict';
function splitSubs(narration, maxLen = 18) {
  const clauses = narration.match(/[^。！？；，、：]+[。！？；，、：]?/g) || [];
  const lines = []; let cur = '';
  for (const cl of clauses) {
    if (cl.length > maxLen) {
      if (cur) { lines.push(cur); cur = ''; }
      for (let i = 0; i < cl.length; i += maxLen) lines.push(cl.slice(i, i + maxLen));
      continue;
    }
    if ((cur + cl).length > maxLen) { if (cur) lines.push(cur); cur = cl; }
    else cur += cl;
  }
  if (cur) lines.push(cur);
  return lines.length ? lines : [narration];
}
function normalizeText(s) { return String(s).replace(/[，。！？；、：·…—\s]/g, ''); }
function checkSubs(override, narration) {
  const a = normalizeText(override.join(''));
  const b = normalizeText(narration);
  if (a === b) return { ok: true, detail: '' };
  let i = 0; while (i < Math.min(a.length, b.length) && a[i] === b[i]) i++;
  return { ok: false, detail: `首个分歧在第 ${i} 字: 字幕="${a.slice(i, i + 10)}" 旁白="${b.slice(i, i + 10)}"` };
}
module.exports = { splitSubs, normalizeText, checkSubs };
```

```js
const PRE = {
  rise: p => ({ op: p, ty: (1 - p) * 48 }),
  fade: p => ({ op: p }),
  scale: p => ({ op: p, sc: 0.9 + 0.1 * p }),
  popLeft: p => ({ op: p, tx: (1 - p) * -40 }),
  grow: p => ({ op: p, sx: p }),
  pulse: p => ({ sc: 1 + 0.06 * Math.sin(p * Math.PI) }),
  highlight: p => ({ br: 1 + 0.35 * Math.sin(p * Math.PI) }),
  fadeOut: p => ({ op: 1 - p }),
  sink: p => ({ op: 1 - p, ty: p * 30 })
};
function chan(...cs) {
  const r = { op: 1, tx: 0, ty: 0, sc: 1, sx: 1, br: 1 };
  for (const c of cs) {
    if (c.op != null) r.op *= c.op;
    if (c.tx) r.tx += c.tx; if (c.ty) r.ty += c.ty;
    if (c.sc != null) r.sc *= c.sc; if (c.sx != null) r.sx *= c.sx;
    if (c.br != null) r.br *= c.br;
  }
  return r;
}
function put(e, c) {
  e.style.opacity = Math.max(0, Math.min(1, c.op));
  const moved = c.tx || c.ty || c.sc !== 1 || c.sx !== 1;
  e.style.transform = moved ? 'translate(' + c.tx + 'px,' + c.ty + 'px) scale(' + (c.sc * c.sx) + ',' + c.sc + ')' : 'none';
  e.style.filter = c.br !== 1 ? 'brightness(' + c.br + ')' : 'none';
}
function cueLt(c, sc) {
  if (c.startsWith('lt:')) return parseFloat(c.slice(3));
  const r = parseFloat(c.slice(10));
  return (sc.audioStart - sc.start / TL.fps) + r * sc.dur;
}
function upEl(e, sp, lt, dur, sc) {
  const en = sp.enter ? PRE[sp.enter.preset](eo(seg(lt, sp.enter.at, sp.enter.at + sp.enter.dur))) : {};
  const ex = sp.exit ? PRE[sp.exit.preset](seg(lt, dur - sp.exit.dur, dur)) : {};
  let em = {};

  if (sp.type === 'stat' && e.__num) {
    const at = sp.enter ? sp.enter.at : 0, d = sp.enter ? sp.enter.dur : 1;
    e.__num.textContent = (sp.props.value * eo(seg(lt, at, at + d))).toFixed(sp.props.decimals || 0);
  }
}
```

```js
function emitSceneHtml({ tokens, scenes, components, preRuntime }) {
```

```js
'use strict';
const fs = require('fs'), path = require('path');
const ir = require('./ir');
const { splitSubs, checkSubs } = require('./subs');
const { PRESET_DEFAULT_DUR, PRE_RUNTIME, ENTER_PRESETS, EMPHASIS_PRESETS, EXIT_PRESETS } = require('./presets');
const { emitSceneHtml } = require('./emit');
const { COMPONENTS, LAYOUT_NAMES } = require('./components');

function normalizeScene(sc) {
  const elements = (sc.elements || []).map(el => {
    const n = JSON.parse(JSON.stringify(el));
    if (n.enter) {
      if (!ENTER_PRESETS.includes(n.enter.preset)) throw new Error(`非法 enter 预设: ${n.enter.preset}`);
      n.enter.at = n.enter.at != null ? n.enter.at : 0;
      n.enter.dur = n.enter.dur != null ? n.enter.dur : PRESET_DEFAULT_DUR[n.enter.preset];
    } else n.enter = null;
    n.emphasis = (n.emphasis || []).map(c => {
      if (!EMPHASIS_PRESETS.includes(c.preset)) throw new Error(`非法 emphasis 预设: ${c.preset}`);
      return { preset: c.preset, cue: c.cue, dur: c.dur != null ? c.dur : PRESET_DEFAULT_DUR[c.preset] };
    });
    if (n.exit) {
      if (!EXIT_PRESETS.includes(n.exit.preset)) throw new Error(`非法 exit 预设: ${n.exit.preset}`);
      n.exit.dur = n.exit.dur != null ? n.exit.dur : PRESET_DEFAULT_DUR[n.exit.preset];
    } else n.exit = null;
    return n;
  });
  return Object.assign({}, sc, { elements });
}

function buildScriptJson(project) {
  const m = project.story.meta;
  const v = m.video || {};
  return {
    schema: 2,
    fps: v.fps || 30, width: v.width || 1920, height: v.height || 1080,
    options: Object.assign({ pad: 1.1, leadIn: .45, introLeadIn: .7, introMinSec: 6.5, outroMinSec: 8.5 }, m.options || {}),
    tts: m.tts || { backend: 'local' },
    scenes: project.scenes.map(sc => ({
      id: sc.id, minSec: sc.minSec || 0, narration: sc.narration,
      subs: sc.subsOverride || splitSubs(sc.narration),
      beat: sc.beat, layout: sc.layout, elements: sc.elements, cues: sc.cues || []
    }))
  };
}

function compileProject(dir) {
  const project = ir.loadProject(dir, LAYOUT_NAMES);
  if (project.mode === 'v1') return { mode: 'v1' };
  const errors = [];
  for (const sc of project.scenes) {
    if (sc.custom) errors.push(`scene ${sc.id} 使用了 custom 逃生舱（M2 实现，M1 暂不支持）`);
    if (sc.subsOverride) {
      const c = checkSubs(sc.subsOverride, sc.narration);
      if (!c.ok) errors.push(`scene ${sc.id} 字幕与旁白不一致: ${c.detail}`);
    }
  }
  let scenes, lint = [];
  try { scenes = project.scenes.map(normalizeScene); }
  catch (e) { errors.push(e.message); scenes = project.scenes; }
  // Task 8 接入 motion-lint；此前 lint 恒为 []
  if (errors.length) return { mode: 'v2', lint, errors };
  const script = buildScriptJson(Object.assign({}, project, { scenes }));
  const html = emitSceneHtml({ tokens: project.tokens, scenes, components: COMPONENTS, preRuntime: PRE_RUNTIME });
  const scriptPath = path.join(dir, 'script.json');
  const htmlPath = path.join(dir, 'scene.html');
  fs.writeFileSync(scriptPath, JSON.stringify(script, null, 2));
  fs.writeFileSync(htmlPath, html);
  return { mode: 'v2', scriptPath, htmlPath, lint, errors };
}

module.exports = { normalizeScene, buildScriptJson, compileProject };
```

```js
const MIME = { '.html': 'text/html', '.js': 'text/javascript', '.json': 'application/json', '.png': 'image/png', '.css': 'text/css' };
async function withPage(dir, fn, query = '') {
  const server = http.createServer((req, res) => {
    let p = decodeURIComponent(req.url.split('?')[0]); if (p === '/') p = '/scene.html';
    const f = path.join(dir, p);
    if (!f.startsWith(dir) || !fs.existsSync(f) || fs.statSync(f).isDirectory()) { res.statusCode = 404; return res.end('nf'); }
    res.setHeader('Content-Type', MIME[path.extname(f)] || 'application/octet-stream');
    res.end(fs.readFileSync(f));
  });
  await new Promise(r => server.listen(0, '127.0.0.1', r));
  const port = server.address().port;
  const browser = await loadPW().chromium.launch({ channel: 'chrome', args: ['--force-color-profile=srgb', '--hide-scrollbars'] });
  try {
    const page = await browser.newPage({ viewport: { width: 1920, height: 1080 }, deviceScaleFactor: 1 });
    await page.goto(`http://127.0.0.1:${port}/scene.html${query ? '?' + query : ''}`, { waitUntil: 'load' });
    await page.waitForFunction(() => window.__ready === true || window.__err, { timeout: 30000 });
    const err = await page.evaluate(() => window.__err);
    if (err) throw new Error('INIT: ' + err);
    return await fn(page);
  } finally { await browser.close(); server.close(); }
}
```

```js
'use strict';
function hexToRgb(hex) { const n = parseInt(hex.slice(1), 16); return [(n >> 16) & 255, (n >> 8) & 255, n & 255]; }
function lum(hex) {
  const [r, g, b] = hexToRgb(hex).map(v => { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); });
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}
function contrastRatio(a, b) { const l1 = lum(a), l2 = lum(b); return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05); }
function estDur(sc) { return Math.max(sc.minSec || 0, (sc.narration || '').length / 4.5 + 1.1); }

function lintMotion({ scenes, tokens, components }) {
  const out = [];
  for (const sc of scenes) {
    const comp = components[sc.layout];
    const boxes = comp.boxes(sc);
    const dur = estDur(sc);
    for (const b of boxes) {
      const bottom = b.y + b.h;
      if (bottom > 940) out.push({ level: 'B', code: 'SUBTITLE_ZONE', scene: sc.id, el: b.id, msg: `盒底 ${bottom} > 940（字幕区）` });
      else if (bottom > 900) out.push({ level: 'W', code: 'NEAR_SUBTITLE', scene: sc.id, el: b.id, msg: `盒底 ${bottom} > 900，接近字幕区` });
      if (b.text) {
        const el = sc.elements.find(e => e.id === b.id);
        const at = el && el.enter ? el.enter.at : 0;
        const ex = el && el.exit ? el.exit.dur : 0;
        const dwell = dur - at - ex;
        const need = Math.max(b.text.length * 0.28, 1.2);
        if (dwell < need) out.push({ level: 'B', code: 'TEXT_DWELL', scene: sc.id, el: b.id, msg: `停留 ${dwell.toFixed(2)}s < 需要 ${need.toFixed(2)}s（${b.text.length} 字）` });
        const fg = tokens.colors[b.colorKey || 'ink'];
        const cr = contrastRatio(fg, tokens.colors.bg);
        if (cr < 4.5) out.push({ level: 'B', code: 'CONTRAST', scene: sc.id, el: b.id, msg: `对比度 ${cr.toFixed(2)} < 4.5（${b.colorKey || 'ink'} on bg）` });
      }
    }
    const movers = sc.elements.filter(e => e.enter);
    let peak = 0;
    for (let t = 0; t < dur; t += 0.1) {
      const c = movers.filter(e => t >= e.enter.at && t <= e.enter.at + e.enter.dur).length;
      if (c > peak) peak = c;
    }
    if (peak > 6) out.push({ level: 'W', code: 'DENSITY', scene: sc.id, msg: `同屏运动峰值 ${peak} > 6` });
  }
  return out;
}
module.exports = { lintMotion, contrastRatio, estDur };
```

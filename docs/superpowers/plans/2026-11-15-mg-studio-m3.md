> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建成 mg-studio 的增量渲染与 QA：scene 级帧缓存（换配音只重渲时长变化的 scene）+ assemble_frames 硬链接拼接 + auto_qa（阻断/警告分级）+ contact_sheet 关键帧人工门。

**Architecture:** 把渲染产物从"单目录全局帧"改为"每 scene 一目录 `frames/<id>/f_%04d.png`（局部帧号）" + `frames/all/f_%04d.png`（硬链接拼接后供单次 ffmpeg 编码）。`lib/fcache` 以 visualHash（scene+tokens+组件/编译器版本）+ frameCount（时长）判定失效；`render.js --all` 逐 scene 渲染并跳过缓存命中，`--scene <id>` 单渲，`--no-cache` 兜底。`auto_qa` 在编码后跑阻断/警告分级；`contact_sheet` 用 PIL 拼关键帧。默认 `node render.js`（无参数）保留全局单目录渲染，向后兼容。

**Tech Stack:** Node ≥ 18（v25.6.1），CommonJS，node:test；Playwright + Chrome；ffmpeg/ffprobe（volumedetect/blackdetect/freezedetect）；python3 + PIL 11.3（contact sheet）。本计划不调用百炼。

```js
'use strict';
const fs = require('fs'), path = require('path'), crypto = require('crypto');
const INDEX = 'index.json';
const CACHE_DIR = ['frames', 'cache'];
function sha(s) { return crypto.createHash('sha256').update(s).digest('hex').slice(0, 16); }
function hashFile(p) { try { return sha(fs.readFileSync(p)); } catch (e) { return null; } }
function visualHash({ scene, tokens, compVersion = '1', compilerVersion = '1' }) {
  // 只取影响画布的 token 子集，避免无关字段抖动导致无谓重渲
  const tk = tokens && tokens.colors ? { colors: tokens.colors, font: tokens.font, subtitle: tokens.subtitle } : tokens;
  return sha(JSON.stringify({ scene, tokens: tk, compVersion, compilerVersion }));
}
function loadIndex(dir) {
  try { return JSON.parse(fs.readFileSync(path.join(dir, ...CACHE_DIR, INDEX), 'utf8')); } catch (e) { return {}; }
}
function saveIndex(dir, idx) {
  const d = path.join(dir, ...CACHE_DIR); fs.mkdirSync(d, { recursive: true });
  fs.writeFileSync(path.join(d, INDEX), JSON.stringify(idx, null, 2));
}
function decide({ visualHash, audioHash, frameCount, cached }) {
  if (!cached) return { render: true, reason: '新增场景' };
  if (cached.visualHash !== visualHash) return { render: true, reason: '视觉变化' };
  if (cached.frameCount !== frameCount) return { render: true, reason: '时长/帧数变化' };
  return { render: false, reason: '缓存命中' };
}
module.exports = { sha, hashFile, visualHash, loadIndex, saveIndex, decide, INDEX, CACHE_DIR };
```

```js
#!/usr/bin/env node
'use strict';
/* assemble_frames.js — 按 timeline 顺序把各 scene 帧硬链接拼到 frames/all/f_%04d.png
 * 用法: node assemble_frames.js [项目目录]
 */
const fs = require('fs'), path = require('path');
function assemble({ dir, timeline }) {
  const all = path.join(dir, 'frames', 'all');
  fs.rmSync(all, { recursive: true, force: true }); fs.mkdirSync(all, { recursive: true });
  let g = 0; const missing = [];
  for (const sc of timeline.scenes) {
    const sd = path.join(dir, 'frames', sc.id);
    const n = sc.end - sc.start;
    for (let k = 0; k < n; k++) {
      const src = path.join(sd, `f_${String(k).padStart(4, '0')}.png`);
      if (!fs.existsSync(src)) { missing.push(`${sc.id}/${k}`); continue; }
      const dst = path.join(all, `f_${String(g).padStart(4, '0')}.png`);
      try { fs.linkSync(src, dst); } catch (e) { fs.copyFileSync(src, dst); }
      g++;
    }
  }
  return { count: g, missing };
}
if (require.main === module) {
  const dir = process.argv[2] ? path.resolve(process.argv[2]) : process.cwd();
  const tl = JSON.parse(fs.readFileSync(path.join(dir, 'timeline.json'), 'utf8'));
  const r = assemble({ dir, timeline: tl });
  console.log('assemble:', r.count, '帧', r.missing.length ? `| 缺 ${r.missing.length} 帧` : '');
  if (r.missing.length) { console.error('缺失样例:', r.missing.slice(0, 5).join(', ')); process.exit(1); }
}
module.exports = { assemble };
```

```js
#!/usr/bin/env node
'use strict';
/* auto_qa.js — G4 QA 门（阻断/警告分级，§9.3）
 * 用法: node auto_qa.js [项目目录] [--skip-media]
 * 产物: reports/qa.json + reports/qa.md；退出码 0 通过 / 1 有 B
 */
const fs = require('fs'), path = require('path');
const { execFileSync } = require('child_process');
const { lintMedical } = require('../lib/medical-lint');
const { checkSubs } = require('../lib/subs');

function parseMeanVolume(s) { const m = /mean_volume:\s*(-?\d+(?:\.\d+)?)\s*dB/.exec(s || ''); return m ? parseFloat(m[1]) : null; }

function ffmpegParse(args) { try { return execFileSync('ffmpeg', args, { stdio: ['ignore','ignore','pipe'] }).toString(); } catch (e) { return (e.stderr||'').toString(); } }

function runQA({ dir, timeline, story, videoPath, skipMedia }) {
  const diag = [];
  // B: 帧数
  const allDir = path.join(dir, 'frames', 'all');
  const n = fs.existsSync(allDir) ? fs.readdirSync(allDir).filter(f => f.endsWith('.png')).length : 0;
  if (n !== timeline.totalFrames) diag.push({ level:'B', code:'QA_FRAME_COUNT', scene:'', msg:`拼接帧 ${n} != 时间轴 ${timeline.totalFrames}` });
  // B: render-status
  try { const st = JSON.parse(fs.readFileSync(path.join(dir,'frames','render-status.json'),'utf8')); if (st.runtimeError) diag.push({ level:'B', code:'QA_RUNTIME', scene:'', msg:'渲染 runtime error: '+st.runtimeError }); } catch (e) {}
  // B: 字幕 override 一致性
  for (const sc of (timeline.scenes||[])) {
    const beat = (story && story.beats || []).find(b => b.id === sc.id);
    const ov = beat && beat._subsOverride;
    if (Array.isArray(ov)) { const c = checkSubs(ov, beat.narration||''); if (!c.ok) diag.push({ level:'B', code:'QA_SUBS_MISMATCH', scene: sc.id, msg: c.detail }); }
  }
  // B: 患者向医学门
  for (const d of lintMedical({ story, scenes: [] })) if (d.level==='B') diag.push({ level:'B', code:d.code, scene:d.scene, msg:d.msg });
  // W: 媒体检查
  if (!skipMedia && videoPath && fs.existsSync(videoPath)) {
    const bl = ffmpegParse(['-hide_banner','-i',videoPath,'-vf','blackdetect=d=0.5:pix_th=0.02','-an','-f','null','-']);
    const black = (bl.match(/black_start:/g)||[]).length;
    if (black) diag.push({ level:'W', code:'QA_BLACK', scene:'', msg:`检出 ${black} 段黑场（核对是否意外）` });
    const fr = ffmpegParse(['-hide_banner','-i',videoPath,'-vf','freezedetect=n=0.001:d=2','-an','-f','null','-']);
    const fz = (fr.match(/freeze_start:/g)||[]).length;
    if (fz) diag.push({ level:'W', code:'QA_FREEZE', scene:'', msg:`检出 ${fz} 段静止 >2s（片头尾可豁免）` });
    const dur = parseFloat((execFileSync('ffprobe',['-v','error','-show_entries','format=duration','-of','csv=p=0',videoPath]).toString()||'0').trim());
    const expect = timeline.totalFrames / (timeline.fps||30);
    if (Math.abs(dur - expect) > 0.3) diag.push({ level:'W', code:'QA_DURATION', scene:'', msg:`时长 ${dur.toFixed(2)}s vs 预期 ${expect.toFixed(2)}s` });
  }
  const B = diag.filter(d=>d.level==='B'), W = diag.filter(d=>d.level==='W');
  return { diagnostics: diag, blocking: B, warnings: W };
}

function main() {
  const args = process.argv.slice(2); const pos = args.filter(a=>!a.startsWith('--'));
  const dir = pos.length ? path.resolve(pos[0]) : process.cwd();
  const skipMedia = args.includes('--skip-media');
  const tl = JSON.parse(fs.readFileSync(path.join(dir,'timeline.json'),'utf8'));
  let story = { meta:{}, beats:[] }; try { const ir=require('../lib/ir'); const {LAYOUT_NAMES}=require('../lib/components'); story = ir.loadProject(dir,LAYOUT_NAMES).story||story; } catch(e){}
  const videoPath = path.join(dir,'out','video_only.mp4');
  const r = runQA({ dir, timeline: tl, story, videoPath, skipMedia });
  const lines = ['# QA 报告', '', `阻断 ${r.blocking.length} · 警告 ${r.warnings.length}`, ''];
  for (const d of r.diagnostics) lines.push(`- [${d.level}] ${d.code} ${d.scene||''} ${d.msg}`);
  fs.mkdirSync(path.join(dir,'reports'),{recursive:true});
  fs.writeFileSync(path.join(dir,'reports','qa.json'), JSON.stringify(r,null,2));
  fs.writeFileSync(path.join(dir,'reports','qa.md'), lines.join('\n')+'\n');
  for (const d of r.diagnostics) console.log(`[${d.level}] ${d.code} ${d.scene||''} ${d.msg}`);
  console.log(`\nQA ${r.blocking.length?'BLOCKED':'OK'}（B=${r.blocking.length} W=${r.warnings.length}）→ reports/qa.md`);
  return r.blocking.length?1:0;
}
if (require.main === module) process.exit(main());
```
